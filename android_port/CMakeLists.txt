cmake_minimum_required(VERSION 3.22)
project(dkc2_android LANGUAGES C)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use Android NDK with ANDROID_ABI=arm64-v8a.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY JOB_POOLS dkc2_compile=2 dkc2_link=1)
set(CMAKE_JOB_POOL_COMPILE dkc2_compile)
set(CMAKE_JOB_POOL_LINK dkc2_link)
get_filename_component(DKC2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SNESRECOMP_ROOT "${DKC2_ROOT}/snesrecomp")
set(SDL2_ROOT "${DKC2_ROOT}/android/third_party/SDL2")
set(GEN "${DKC2_ROOT}/generated/snesrecomp")
option(DKC2_ANDROID_SOURCE_CHECK "Compile host objects ONLY; never produces an APK or runnable game" OFF)
foreach(required "${SNESRECOMP_ROOT}/runner/runner.cmake" "${SDL2_ROOT}/CMakeLists.txt")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing ${required}; prepare pinned dependencies first.")
  endif()
endforeach()
file(GLOB GEN_C CONFIGURE_DEPENDS "${GEN}/*.c")
if(NOT DKC2_ANDROID_SOURCE_CHECK)
  if(NOT GEN_C OR NOT EXISTS "${GEN}/android-generation.json")
    message(FATAL_ERROR "Missing verified generated DKC2 game sources. Run python android/tools/build_android.py --rom <your-ROM> --prepare-only. No placeholder game is built.")
  endif()
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${SDL2_ROOT}" "${CMAKE_BINARY_DIR}/SDL2")
set(SNESRECOMP_SDL_BACKEND SDL2 CACHE STRING "" FORCE)
set(SNESRECOMP_ENABLE_TRACE OFF CACHE BOOL "" FORCE)
set(SNES_COSIM OFF CACHE BOOL "" FORCE)
set(SNESRECOMP_ENABLE_MODS OFF CACHE BOOL "" FORCE)
set(SNESRECOMP_ENABLE_FRAME_FINGERPRINTS OFF CACHE BOOL "" FORCE)
set(SNESRECOMP_ENABLE_PPU_DMA_HISTORY OFF CACHE BOOL "" FORCE)
set(SNESRECOMP_ENABLE_DISPATCH_HISTORY OFF CACHE BOOL "" FORCE)
include("${SNESRECOMP_ROOT}/runner/runner.cmake")
set(ENGINE ${SNESRECOMP_RUNNER_SOURCES})
list(REMOVE_ITEM ENGINE "${SNESRECOMP_ROOT}/runner/src/keybinds.c" "${SNESRECOMP_ROOT}/runner/src/host_report.c")
set(GAME dkc2_game.c dkc2_video.c headless_host.c input_playback.c verified_rom.c)
list(TRANSFORM GAME PREPEND "${DKC2_ROOT}/runner/")
set(HOST android_host.c android_platform.c desktop_audio_rate.c desktop_filter.c)
if(DKC2_ANDROID_SOURCE_CHECK)
  # This target validates C/NDK interfaces without manufacturing cartridge code.
  add_library(android_host_check OBJECT ${HOST} ${GAME})
  set(PORT_TARGET android_host_check)
else()
  add_library(main SHARED ${ENGINE} ${GEN_C} ${GAME} ${HOST})
  set(PORT_TARGET main)
endif()
target_include_directories(${PORT_TARGET} PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${DKC2_ROOT}/runner" "${DKC2_ROOT}/recomp" ${SNESRECOMP_RUNNER_INCLUDE_DIRS})
target_compile_definitions(${PORT_TARGET} PRIVATE SNESRECOMP_SDL3=0 SNESRECOMP_TRACE=0 SNESRECOMP_REVERSE_DEBUG=0 SNESRECOMP_EXTERNAL_RAM_ROUTINE_GUARDS=1 SYSTEM_VOLUME_MIXER_AVAILABLE=0 _POSIX_C_SOURCE=200809L DKC2_BUILD_COMMIT="e181419f" DKC2_BUILD_CONFIG="android-0.1.0-dev")
target_compile_options(${PORT_TARGET} PRIVATE -fno-strict-aliasing -Werror=implicit-function-declaration)
target_link_libraries(${PORT_TARGET} PRIVATE SDL2::SDL2 ${SNESRECOMP_RUNNER_LIBRARIES} log android m)
if(NOT DKC2_ANDROID_SOURCE_CHECK)
  if(TARGET SDL2main)
    target_link_libraries(main PRIVATE SDL2main)
  endif()
  target_link_options(main PRIVATE "-Wl,--no-undefined")
  foreach(t main SDL2)
    if(TARGET ${t})
      target_link_options(${t} PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
    endif()
  endforeach()
endif()
